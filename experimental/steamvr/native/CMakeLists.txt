cmake_minimum_required(VERSION 3.21)
project(VRizationSteamVRNative VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Native preview requires Windows x64 MSVC and its Windows SDK. No runtime is installed or registered by this build.")
endif()
if(NOT OPENVR_SDK_ROOT)
    message(FATAL_ERROR "Use scripts/build_steamvr_native.py to fetch/verify the pinned OpenVR SDK first.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
add_compile_options(/W4 /permissive- /utf-8)
add_library(driver_vrization_phone SHARED phone_driver.cpp)
target_include_directories(driver_vrization_phone PRIVATE "${OPENVR_SDK_ROOT}/headers")
set_target_properties(driver_vrization_phone PROPERTIES PREFIX "" OUTPUT_NAME "driver_vrization_phone")
foreach(program Mirror Overlay)
    string(TOLOWER "${program}" source)
    add_executable(VRization-SteamVR-${program} ${source}.cpp)
    target_include_directories(VRization-SteamVR-${program} PRIVATE "${OPENVR_SDK_ROOT}/headers")
    target_link_libraries(VRization-SteamVR-${program} PRIVATE "${OPENVR_SDK_ROOT}/lib/win64/openvr_api.lib" d3d11 dxgi d3dcompiler)
    add_custom_command(TARGET VRization-SteamVR-${program} POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${OPENVR_SDK_ROOT}/bin/win64/openvr_api.dll" "$<TARGET_FILE_DIR:VRization-SteamVR-${program}>/openvr_api.dll")
endforeach()
include(CTest)
if(BUILD_TESTING)
    add_executable(VRization-SteamVR-NativeTests native_tests.cpp)
    target_include_directories(VRization-SteamVR-NativeTests PRIVATE "${OPENVR_SDK_ROOT}/headers")
    target_link_libraries(VRization-SteamVR-NativeTests PRIVATE d3d11 dxgi d3dcompiler)
    add_dependencies(VRization-SteamVR-NativeTests driver_vrization_phone)
    add_test(NAME native-ipc-pose-warp-factory COMMAND VRization-SteamVR-NativeTests "$<TARGET_FILE:driver_vrization_phone>")
    set_tests_properties(native-ipc-pose-warp-factory PROPERTIES TIMEOUT 60)
endif()
install(TARGETS VRization-SteamVR-Mirror VRization-SteamVR-Overlay RUNTIME DESTINATION native)
install(FILES "${OPENVR_SDK_ROOT}/bin/win64/openvr_api.dll" DESTINATION native)
install(TARGETS driver_vrization_phone RUNTIME DESTINATION drivers/vrization_phone/bin/win64)
install(FILES driver.vrdrivermanifest DESTINATION drivers/vrization_phone)
install(DIRECTORY resources/ DESTINATION drivers/vrization_phone/resources)
install(FILES README.md DESTINATION native)
install(FILES licenses/README.md licenses/OpenVR-LICENSE.txt DESTINATION native/licenses)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../../../LICENSE" DESTINATION native/licenses RENAME VRization-MIT.txt)
